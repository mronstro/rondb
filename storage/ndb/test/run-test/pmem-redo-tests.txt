# RONDB-1150: the autotest cases run before each push of PMEM code, in a
# build with error inserts, on the .2node cluster of conf-pmem-nodax.cnf
# (and of conf-pmem-emulated.cnf on memmap= hosts), and with the
# power-loss emulation on .2node_powerloss. GciLedgerPowerLoss is skipped
# without the emulation. The crash error inserts of a restart act in node
# restarts here, as StopOnError=1; the MTR test
# ndb_redo_pmem_ledger_crash runs them in system restarts.
#

# System restarts (recovery plan A1)
cmd: testSystemRestart
args: -n SR1 T1
max-time: 4140

cmd: testSystemRestart
args: -n SR2 T1
max-time: 3240

# Node restarts (A2). Not GCPSaveAndFileChange: its error insert 5114
# delays the last write in a REDO file of file mode, which the PMEM
# format does not make, so the GCP save never stalls as the case needs.
# PmemGcpSaveAndFileChange checks that the GCP saves go on instead (M6).
cmd: testNodeRestart
args: -n NoLoad T1
max-time: 1080

cmd: testNodeRestart
args: -n PmemGcpSaveAndFileChange --records=2000 T3
max-time: 900

# The REDO log (A3)
cmd: testRedo
args: -n RestartOK T1
max-time: 1440

cmd: testRedo
args: -n RestartFDSR T1
max-time: 1440

cmd: testSystemRestart
args: -n MaxRedoRecordSR --records=200 T1
max-time: 1800

# The mode switch (A5). With StopOnError=1 the refused restarts are not
# checked here; the MTR test ndb_redo_pmem_mode_switch checks them.
cmd: testSystemRestart
args: -n PmemRedoModeSwitch --records=1000 T1
max-time: 3600
